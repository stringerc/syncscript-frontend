// Standard manager integration pattern
interface ManagerIntegration {
  name: string;
  version: string;
  dependencies: string[];
  initialize(): Promise<void>;
  destroy(): Promise<void>;
  getStatus(): ManagerStatus;
}

class ManagerRegistry {
  private managers: Map<string, ManagerIntegration> = new Map();
  
  register(manager: ManagerIntegration): void;
  initialize(): Promise<void>;
  getManager<T>(name: string): T;
  getStatus(): RegistryStatus;
}
